```dart
// ✅ 重构后：清晰的BLoC架构
class GoalPage extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return BlocBuilder<GoalBloc, GoalState>(
      builder: (context, state) {
        if (state is GoalLoading) {
          return const LoadingWidget();
        }
        
        if (state is GoalError) {
          return ErrorWidget(state.message);
        }
        
        if (state is GoalsLoaded) {
          return _buildGoalContent(context, state);
        }
        
        return const SizedBox.shrink();
      },
    );
  }

  Widget _buildGoalContent(BuildContext context, GoalsLoaded state) {
    return Column(
      children: [
        // UI完全由状态驱动，无需setState
        GoalTreeViewBloc(
          goals: state.goals,
          currentGoal: state.currentGoal,
          viewMode: state.viewMode,
          onGoalTap: (goal) => context.read<GoalBloc>().add(SelectGoal(goal)),
          onEditGoal: (goal) => context.read<GoalBloc>().add(UpdateGoal(goal)),
        ),
      ],
    );
  }
}
```

## ⚡ 性能优化机制

### 1. UI批量更新优化

```dart
// ✅ 智能UI批量更新机制
class GoalBloc extends Bloc<GoalEvent, GoalState> {
  late final UIBatchUpdater _uiBatchUpdater;
  
  GoalBloc({required this.repository}) : super(GoalInitial()) {
    _uiBatchUpdater = UIBatchUpdater(
      config: UIBatchUpdateConfig.highPerformance(),
    );
  }

  /// 智能批量emit - 根据状态类型和优先级选择最佳更新策略
  void _smartEmit(GoalState newState, Emitter<GoalState> emit,
      {UIUpdatePriority priority = UIUpdatePriority.normal}) {
    
    // 错误状态和加载状态立即发射
    if (newState is GoalError || newState is GoalLoading) {
      emit(newState);
      return;
    }

    // 根据状态类型选择更新策略
    if (newState is GoalsLoaded) {
      final updateType = _determineUpdateType(newState);
      final updatePriority = _determineUpdatePriority(newState, priority);
      
      _batchedEmits++;
      _uiBatchUpdater.addUpdate(
        _createUIUpdateItem(newState, updateType, updatePriority, emit),
      );
    }
  }

  // 使用示例
  void _onSelectGoal(SelectGoal event, Emitter<GoalState> emit) {
    if (state is GoalsLoaded) {
      final currentState = state as GoalsLoaded;
      // 目标选择使用立即优先级，确保用户交互响应及时
      _smartEmit(currentState.copyWith(
        currentGoal: event.goal,
      ), emit, priority: UIUpdatePriority.immediate);
    }
  }
}
```

### 2. 数据库批量操作优化

```dart
// ✅ 数据库批量操作优化
class GoalRepositoryImpl implements GoalRepository {
  late final DatabaseBatchOptimizer _batchOptimizer;
  
  GoalRepositoryImpl(this._dbHelper) {
    _batchOptimizer = DatabaseBatchOptimizer(
      databaseHelper: _dbHelper,
      config: DatabaseBatchConfig.highPerformance(),
    );
  }

  @override
  Future<List<int>> batchInsertGoals(List<Goal> goals) async {
    final result = await _batchOptimizer.batchInsertGoals(goals);
    return List.generate(result.successfulItems, (index) => index + 1);
  }

  /// 智能批量操作 - 根据操作类型和数据量选择最佳策略
  Future<List<int>> smartBatchOperation(
    DatabaseOperationType type,
    List<dynamic> items,
  ) async {
    switch (type) {
      case DatabaseOperationType.insert:
        final goals = items.cast<Goal>();
        return await batchInsertGoals(goals);
      case DatabaseOperationType.update:
        final goals = items.cast<Goal>();
        return await batchUpdateGoals(goals);
      case DatabaseOperationType.delete:
        final goalIds = items.cast<int>();
        return await batchDeleteGoals(goalIds);
    }
  }

  /// 事务性批量同步操作
  Future<Map<String, dynamic>> batchSyncGoals({
    List<Goal>? goalsToInsert,
    List<Goal>? goalsToUpdate,
    List<int>? goalIdsToDelete,
  }) async {
    return await executeInTransaction(() async {
      final results = <String, dynamic>{};
      
      // 批量删除 → 批量更新 → 批量插入
      if (goalIdsToDelete?.isNotEmpty == true) {
        final deleteResults = await batchDeleteGoals(goalIdsToDelete!);
        results['deleted'] = deleteResults.length;
      }
      
      if (goalsToUpdate?.isNotEmpty == true) {
        final updateResults = await batchUpdateGoals(goalsToUpdate!);
        results['updated'] = updateResults.length;
      }
      
      if (goalsToInsert?.isNotEmpty == true) {
        final insertResults = await batchInsertGoals(goalsToInsert!);
        results['inserted'] = insertResults.length;
      }
      
      results['timestamp'] = DateTime.now().millisecondsSinceEpoch;
      return results;
    });
  }
}
```
